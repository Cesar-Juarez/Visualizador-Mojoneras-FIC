# -*- coding: utf-8 -*- from google.colab import drive, files import pandas as pd import folium import os import html from urllib.parse import quote # 1. CONECTAR GOOGLE DRIVE drive.mount('/content/drive') # 2. CONFIGURACIÓN ruta_archivo = '/content/drive/MyDrive/Mojoneras_Fic.csv' # Columnas requeridas columnas_requeridas = [ 'Mojonera', 'Elevacion', 'E', 'N', 'X', 'Y', 'Latitud', 'Longitud' ] def generar_visualizador_mojoneras(): if not os.path.exists(ruta_archivo): print("No se encontró el archivo:", ruta_archivo) return None try: # 3. LEER CSV df = pd.read_csv(ruta_archivo, encoding='utf-8-sig') df.columns = df.columns.str.strip() faltantes = [ c for c in columnas_requeridas if c not in df.columns ] if faltantes: print("Faltan estas columnas:", faltantes) print("Columnas disponibles:", list(df.columns)) return None # 4. LIMPIAR COORDENADAS for columna in ['Latitud', 'Longitud', 'Elevacion', 'E', 'N', 'X', 'Y']: df[columna] = pd.to_numeric( df[columna], errors='coerce' ) df = df.dropna(subset=['Latitud', 'Longitud']) if df.empty: print("No existen coordenadas geográficas válidas.") return None # 5. CREAR MAPA centro = [ df['Latitud'].mean(), df['Longitud'].mean() ] m = folium.Map( location=centro, zoom_start=19, max_zoom=25, control_scale=True ) # Capa satelital folium.TileLayer( tiles='https://mt1.google.com/vt/lyrs=y&x={x}&y={y}&z={z}', attr='Google Satellite', name='Imagen satelital', overlay=False, control=True ).add_to(m) # Capa de calles folium.TileLayer( 'OpenStreetMap', name='Mapa de calles', control=True ).add_to(m) # 6. GRUPO DE MOJONERAS grupo = folium.FeatureGroup( name='Mojoneras registradas', show=True ) coordenadas = [] # 7. AGREGAR MOJONERAS for _, fila in df.iterrows(): nombre = str(fila['Mojonera']) nombre_html = html.escape(nombre) lat = float(fila['Latitud']) lon = float(fila['Longitud']) coordenadas.append([lat, lon]) def mostrar(valor, decimales=None): if pd.isna(valor): return 'No disponible' if decimales is not None: return f"{float(valor):,.{decimales}f}" return html.escape(str(valor)) elevacion = mostrar(fila['Elevacion'], 3) este = mostrar(fila['E'], 3) norte = mostrar(fila['N'], 3) x = mostrar(fila['X'], 3) y = mostrar(fila['Y'], 3) latitud = mostrar(fila['Latitud'], 8) longitud = mostrar(fila['Longitud'], 8) # 8. VENTANA DE INFORMACIÓN popup_html = f"""
REGISTRO TOPOGRÁFICO
{nombre_html}
INFORMACIÓN GENERAL
Elevación {elevacion} m
COORDENADAS UTM
Este (E) {este}
Norte (N) {norte}
COORDENADAS TOPOGRÁFICAS
X {x}
Y {y}
COORDENADAS GEOGRÁFICAS
Latitud {latitud}
Longitud {longitud}
ABRIR CÉDULA PDF
""" # 9. MARCADOR PERSONALIZADO icono_html = f"""
{nombre_html}
""" folium.Marker( location=[lat, lon], popup=folium.Popup( popup_html, max_width=330 ), tooltip=nombre_html, icon=folium.DivIcon( html=icono_html, icon_size=(120, 35), icon_anchor=(60, 35) ) ).add_to(grupo) grupo.add_to(m) # 10. AJUSTAR ZOOM A TODAS LAS MOJONERAS if len(coordenadas) == 1: m.location = coordenadas[0] m.zoom_start = 20 else: m.fit_bounds( coordenadas, padding=[45, 45], max_zoom=20 ) # 11. CONTROL DE CAPAS folium.LayerControl( collapsed=False ).add_to(m) # 12. GUARDAR VISUALIZADOR nombre_salida = 'index.html' m.save(nombre_salida) print("VISUALIZADOR GENERADO CORRECTAMENTE") print("Total de mojoneras:", len(df)) print("Archivo:", nombre_salida) return m except Exception as e: print("Error durante el proceso:", e) return None # 13. EJECUTAR mapa = generar_visualizador_mojoneras() # 14. DESCARGAR HTML if mapa is not None: files.download('index.html')
